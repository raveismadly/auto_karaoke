# Копируем файлы в выходную директорию
import os
import shutil

output_dir = '/home/user/output'
if not os.path.exists(output_dir):
    os.makedirs(output_dir)

# Копируем все файлы в выходную директорию
for file in ['requirements.txt', 'main.py', 'audio_processor.py', 'text_recognizer.py', 'karaoke_generator.py', 'utils.py']:
    shutil.copy(file, os.path.join(output_dir, file))

print("All files copied to output directory")
